WITH orders_2024 AS (
  SELECT order_id, customer_id, total_amount, status
  FROM orders
  WHERE order_date >= '2024-01-01' AND order_date < '2025-01-01'
)
SELECT order_id, customer_id, total_amount
FROM orders_2024
WHERE status = 'DELIVERED'
ORDER BY order_id
